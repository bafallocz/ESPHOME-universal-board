  # --------------------------------------------------------------------------

        # --- 1. Initialize AD5593R DAC Analog Outputs ---

            id: dac_out_0

        - output.set_level:
            id: dac_out_1
            level: 0.70

        # --- 2. Initialize 4x High-Power MOSFET PWM Channels (1 kHz) ---
        # PWM 1 (GPIO8  -> TC4427 IN_A -> FET 1): 25% Duty Cycle
        - output.set_level:
            id: pwm_1_output
            level: 0.25
        # PWM 2 (PCA9685 CH0 -> TC4427 IN_B -> FET 2): 50% Duty Cycle
        - output.set_level:
            id: pwm_2_output
            level: 0.50
        # PWM 3 (GPIO9  -> TC4427 IN_A -> FET 3): 75% Duty Cycle
        - output.set_level:
            id: pwm_3_output
            level: 0.75
        # PWM 4 (GPIO10 -> TC4427 IN_B -> FET 4): 90% Duty Cycle
        - output.set_level:
            id: pwm_4_output
            level: 0.90

        # --- 3. Initialize PCA9685 Expansion Header Channels (EX_PWM1..15) ---
        # Set staggered duty cycles so every pin on JP1 header shows distinct activity
        - output.set_level: { id: out_ex_pwm1,  level: 0.10 }

        - output.set_level: { id: out_ex_pwm6,  level: 0.60 }
        - output.set_level: { id: out_ex_pwm7,  level: 0.70 }
        - output.set_level: { id: out_ex_pwm8,  level: 0.80 }
        - output.set_level: { id: out_ex_pwm9,  level: 0.90 }
        - output.set_level: { id: out_ex_pwm10, level: 0.25 }
        - output.set_level: { id: out_ex_pwm11, level: 0.50 }
        - output.set_level: { id: out_ex_pwm12, level: 0.75 }

        - output.set_level: { id: out_ex_pwm14, level: 0.66 }
